cmake_minimum_required(VERSION 3.20)
project(WallpaperLab VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(ENGINE_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/build-engine" CACHE PATH "Built engine tree")
file(GLOB CEF_ROOT "${ENGINE_BUILD}/cef/cef_binary_*_linux64_minimal")
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Build the engine first; see scripts/build.sh")
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(USE_EXTERNAL_GLFW ON CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/raylib)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
add_library(engine SHARED IMPORTED)
set_target_properties(engine PROPERTIES IMPORTED_LOCATION "${ENGINE_BUILD}/output/liblinux-wallpaperengine-lib.so")
add_executable(wallpaper-lab src/app.cpp src/catalog.cpp src/engine_entry.cpp src/selector.c src/gif.c)
target_compile_definitions(wallpaper-lab PRIVATE _GNU_SOURCE)
target_include_directories(wallpaper-lab PRIVATE src vendor/engine/src vendor/engine/src/External/kissfft vendor/engine/src/External/json/include ${CEF_ROOT} ${SDL2_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/include")
target_link_libraries(wallpaper-lab PRIVATE raylib engine pthread)
target_link_options(wallpaper-lab PRIVATE "-Wl,-rpath-link,${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib" "-Wl,-rpath-link,${ENGINE_BUILD}/output")
set_target_properties(wallpaper-lab PROPERTIES BUILD_RPATH "${ENGINE_BUILD}/output;${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib" INSTALL_RPATH "$ORIGIN;$ORIGIN/lib;$ORIGIN/deps")
enable_testing()
add_executable(catalog-test tests/catalog_test.cpp src/catalog.cpp)
target_include_directories(catalog-test PRIVATE src vendor/engine/src/External/json/include)
add_test(NAME catalog COMMAND catalog-test)
add_executable(gif-test tests/gif_test.c src/gif.c)
target_include_directories(gif-test PRIVATE src)
target_link_libraries(gif-test PRIVATE m)
add_test(NAME gif-timing COMMAND gif-test)

install(TARGETS wallpaper-lab RUNTIME DESTINATION . COMPONENT app)
