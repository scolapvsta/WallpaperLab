cmake_minimum_required(VERSION 3.20)
project(WallpaperLab VERSION 0.13.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(ENGINE_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/build-engine" CACHE PATH "Built engine tree")
file(GLOB CEF_ROOT "${ENGINE_BUILD}/cef/cef_binary_*_linux64_minimal")
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Build the engine first; see scripts/build.sh")
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(USE_EXTERNAL_GLFW ON CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/raylib)
find_package(PkgConfig REQUIRED)
find_package(Freetype REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(PANGO REQUIRED IMPORTED_TARGET pangocairo)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(TRANSLATIONS_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/translations_data.h")
add_custom_command(OUTPUT "${TRANSLATIONS_HEADER}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate-translations.py" "${CMAKE_CURRENT_SOURCE_DIR}/src/translations.json" "${TRANSLATIONS_HEADER}"
    DEPENDS src/translations.json scripts/generate-translations.py VERBATIM)
pkg_check_modules(SDL2 REQUIRED sdl2)
add_library(engine SHARED IMPORTED)
set_target_properties(engine PROPERTIES IMPORTED_LOCATION "${ENGINE_BUILD}/output/liblinux-wallpaperengine-lib.so")
add_library(cef SHARED IMPORTED)
set_target_properties(cef PROPERTIES IMPORTED_LOCATION "${ENGINE_BUILD}/output/libcef.so")
add_executable(wallpaper-lab "${TRANSLATIONS_HEADER}" src/app.cpp src/workshop.cpp src/i18n.cpp src/displays.cpp src/catalog.cpp src/preferences.cpp src/shortcuts.cpp src/popup.cpp src/engine_entry.cpp src/selector.c src/gif.c)
target_compile_definitions(wallpaper-lab PRIVATE _GNU_SOURCE)
target_include_directories(wallpaper-lab PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" src vendor/engine/src vendor/engine/src/External/kissfft vendor/engine/src/External/json/include ${CEF_ROOT} ${SDL2_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/include")
# Chromium's close() interposer uses RTLD_NEXT. Keep CEF before libc in the
# executable's direct dependencies, even though only the engine calls its API.
target_link_libraries(wallpaper-lab PRIVATE raylib Freetype::Freetype engine "-Wl,--no-as-needed" cef "-Wl,--as-needed" pthread PkgConfig::CURL PkgConfig::PANGO dl)
target_link_options(wallpaper-lab PRIVATE "-Wl,-rpath-link,${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib" "-Wl,-rpath-link,${ENGINE_BUILD}/output")
set_target_properties(wallpaper-lab PROPERTIES BUILD_RPATH "${ENGINE_BUILD}/output;${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib" INSTALL_RPATH "$ORIGIN;$ORIGIN/lib;$ORIGIN/deps")
enable_testing()
add_executable(catalog-test tests/catalog_test.cpp src/catalog.cpp)
target_include_directories(catalog-test PRIVATE src vendor/engine/src/External/json/include)
add_test(NAME catalog COMMAND catalog-test)
add_executable(gif-test tests/gif_test.c src/gif.c)
target_include_directories(gif-test PRIVATE src)
target_link_libraries(gif-test PRIVATE m)
add_test(NAME gif-timing COMMAND gif-test)

install(TARGETS wallpaper-lab RUNTIME DESTINATION . COMPONENT app)

add_executable(preferences-test tests/preferences_test.cpp src/preferences.cpp src/shortcuts.cpp src/catalog.cpp)
target_include_directories(preferences-test PRIVATE src vendor/engine/src/External/json/include)
add_test(NAME preferences COMMAND preferences-test)

add_executable(popup-test tests/popup_test.cpp src/popup.cpp src/catalog.cpp)
target_include_directories(popup-test PRIVATE src vendor/engine/src/External/json/include)
add_test(NAME popup COMMAND popup-test)

add_executable(display-layout-test tests/display_layout_test.cpp src/displays.cpp src/catalog.cpp)
target_include_directories(display-layout-test PRIVATE src vendor/engine/src vendor/engine/src/External/json/include)
add_test(NAME display-layout COMMAND display-layout-test)

add_executable(scene-parser-test tests/scene_parser_test.cpp)
target_include_directories(scene-parser-test PRIVATE vendor/engine/src vendor/engine/src/External/json/include "${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/include")
target_link_libraries(scene-parser-test PRIVATE engine "-Wl,--no-as-needed" cef "-Wl,--as-needed")
target_link_options(scene-parser-test PRIVATE "-Wl,-rpath-link,${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib" "-Wl,-rpath-link,${ENGINE_BUILD}/output")
set_target_properties(scene-parser-test PROPERTIES BUILD_RPATH "${ENGINE_BUILD}/output;${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib")
add_test(NAME scene-parser COMMAND scene-parser-test)
set_tests_properties(scene-parser PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${ENGINE_BUILD}/output:${CMAKE_CURRENT_SOURCE_DIR}/local/sysroot/usr/lib")

add_executable(i18n-test tests/i18n_test.cpp src/i18n.cpp "${TRANSLATIONS_HEADER}")
target_include_directories(i18n-test PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated" vendor/engine/src/External/json/include)
add_test(NAME i18n COMMAND i18n-test)

add_executable(project-properties-test tests/project_properties_test.cpp)
target_include_directories(project-properties-test PRIVATE src vendor/engine/src vendor/engine/src/External/json/include)
add_test(NAME project-properties COMMAND project-properties-test)

add_executable(project-source-test tests/project_source_test.cpp src/catalog.cpp)
target_include_directories(project-source-test PRIVATE src vendor/engine/src/External/json/include)
add_test(NAME project-source COMMAND project-source-test)
